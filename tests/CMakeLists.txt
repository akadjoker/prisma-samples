function(samples_test name)
  cmake_parse_arguments(T "" "" "LIBS;DEFINES" ${ARGN})
  add_executable(test_${name} test_${name}.cpp)
  target_link_libraries(test_${name} PRIVATE ct ${T_LIBS})
  if(T_DEFINES)
    target_compile_definitions(test_${name} PRIVATE ${T_DEFINES})
  endif()
  add_test(NAME ${name} COMMAND test_${name})
endfunction()

find_library(PRISMA_EGL_TESTS EGL)

function(samples_gpu_test name)
  cmake_parse_arguments(T "" "" "LIBS;SHADERS" ${ARGN})
  if(NOT PRISMA_SHADER_TOOLS OR NOT PRISMA_EGL_TESTS)
    return()
  endif()
  add_executable(test_${name} test_${name}.cpp)
  target_link_libraries(test_${name} PRIVATE prisma ct demo_common ${T_LIBS} ${PRISMA_EGL_TESTS})
  prisma_shaders(test_${name} ${T_SHADERS})
  if(PRISMA_VULKAN)
    target_compile_definitions(test_${name} PRIVATE PRISMA_TEST_VULKAN)
    target_link_libraries(test_${name} PRIVATE Vulkan::Vulkan)
  endif()
  add_test(NAME gl_${name} COMMAND test_${name})
  set_tests_properties(gl_${name} PROPERTIES LABELS gpu)
  if(PRISMA_VULKAN)
    add_test(NAME vulkan_${name} COMMAND test_${name} vulkan)
    set_tests_properties(vulkan_${name} PROPERTIES LABELS gpu)
  endif()
endfunction()
